async function initialiseEntraAuthentication() { preserveAccessParameters(); const authStatus = document.getElementById("authStatus"); const authUser = document.getElementById("authUser"); /* Diagnóstico visível no próprio ecrã. Assim não precisamos de F12. */ const diagnostic = document.createElement("div"); diagnostic.id = "authDiagnostic"; diagnostic.style.cssText = ` margin-top:15px; padding:14px; border-radius:10px; background:#070b12; border:1px solid #263246; color:#cbd5e1; font-family:monospace; font-size:13px; line-height:1.6; white-space:pre-wrap; `; diagnostic.innerHTML = "🔎 Diagnóstico de autenticação\n" + "A iniciar..."; if (authStatus && authStatus.parentNode) { authStatus.parentNode.insertBefore( diagnostic, authStatus.nextSibling ); } function diagnosticStep(message, type = "info") { const icons = { info: "🔎", ok: "✅", warning: "⚠️", error: "❌" }; const icon = icons[type] || "🔎"; if (diagnostic) { const current = diagnostic.innerHTML; diagnostic.innerHTML = current + "\n" + icon + " " + escapeHtml(message); } console.log( "[Cyber Challenge Auth]", message ); } function diagnosticError(error) { const message = error && error.message ? error.message : String(error); diagnosticStep( message, "error" ); } try { /* ---------------------------------------------------- 1 — MSAL ---------------------------------------------------- */ diagnosticStep( "1/8 A verificar biblioteca MSAL..." ); if (typeof msal === "undefined") { throw new Error( "A biblioteca MSAL não foi carregada." ); } diagnosticStep( "MSAL carregado.", "ok" ); /* ---------------------------------------------------- 2 — Inicializar MSAL ---------------------------------------------------- */ diagnosticStep( "2/8 A inicializar MSAL..." ); msalInstance = new msal.PublicClientApplication( msalConfig ); diagnosticStep( "MSAL inicializado.", "ok" ); /* ---------------------------------------------------- 3 — Processar retorno do loginRedirect ---------------------------------------------------- */ diagnosticStep( "3/8 A processar retorno do Microsoft Entra..." ); const response = await msalInstance.handleRedirectPromise(); if (response) { diagnosticStep( "O Entra devolveu uma resposta de autenticação.", "ok" ); if (response.account) { entraAccount = response.account; msalInstance.setActiveAccount( entraAccount ); diagnosticStep( "Conta recebida no retorno do Entra.", "ok" ); } if (response.idToken) { entraIdToken = response.idToken; diagnosticStep( "ID Token recebido no retorno.", "ok" ); } } else { diagnosticStep( "Não existe retorno de loginRedirect. A verificar sessão existente." ); } /* ---------------------------------------------------- 4 — Procurar conta existente ---------------------------------------------------- */ diagnosticStep( "4/8 A procurar conta Entra existente..." ); if (!entraAccount) { const accounts = msalInstance.getAllAccounts(); diagnosticStep( `Contas encontradas no MSAL: ${accounts.length}` ); if (accounts.length > 0) { entraAccount = accounts[0]; msalInstance.setActiveAccount( entraAccount ); diagnosticStep( "Conta Entra encontrada.", "ok" ); } } /* ---------------------------------------------------- 5 — Se não houver conta, iniciar login ---------------------------------------------------- */ if (!entraAccount) { diagnosticStep( "5/8 Nenhuma sessão Entra encontrada.", "warning" ); if (authUser) { authUser.textContent = "A iniciar sessão Microsoft..."; } diagnosticStep( "A abrir Microsoft Entra para autenticação..." ); /* * loginRedirect termina a execução desta página * e redireciona para o Microsoft Entra. */ await msalInstance.loginRedirect( loginRequest ); return false; } /* ---------------------------------------------------- 6 — Obter ID Token ---------------------------------------------------- */ diagnosticStep( "5/8 Conta Entra encontrada.", "ok" ); diagnosticStep( "6/8 A verificar ID Token..." ); /* * Se recebemos o token no redirect, usamos esse. * Caso contrário tentamos obtê-lo silenciosamente. */ if (!entraIdToken) { diagnosticStep( "Não existe ID Token em memória. A pedir token silenciosamente..." ); try { const silentResponse = await msalInstance.acquireTokenSilent({ ...loginRequest, account: entraAccount }); entraIdToken = silentResponse.idToken || null; if (entraIdToken) { diagnosticStep( "ID Token obtido com sucesso.", "ok" ); } else { throw new Error( "O Microsoft Entra não devolveu um ID Token." ); } } catch (silentError) { console.error( "Erro acquireTokenSilent:", silentError ); diagnosticStep( "Não foi possível obter o token silenciosamente.", "warning" ); diagnosticStep( "A iniciar nova autenticação Microsoft Entra..." ); if (authUser) { authUser.textContent = "A renovar sessão Microsoft..."; } await msalInstance.loginRedirect( loginRequest ); return false; } } if (!entraIdToken) { throw new Error( "Não foi possível obter um ID Token Entra." ); } /* ---------------------------------------------------- 7 — Mostrar identidade ---------------------------------------------------- */ diagnosticStep( "7/8 A validar identidade do participante..." ); showAuthenticatedUser(); const identity = establishParticipantIdentity(); diagnosticStep( `Identidade: ${identity.name || "Sem nome"}`, "ok" ); diagnosticStep( `Email: ${identity.email}`, "ok" ); diagnosticStep( `Cyber Codename: ${identity.cyberCodename}`, "ok" ); diagnosticStep( `Object ID: ${identity.objectId}`, "ok" ); /* ---------------------------------------------------- 8 — Autenticação concluída ---------------------------------------------------- */ diagnosticStep( "8/8 Autenticação Entra concluída.", "ok" ); if (authStatus) { authStatus.style.borderColor = "#22c55e"; } if (authUser) { /* * Mantemos a identidade normal no topo. */ showAuthenticatedUser(); } return true; } catch (error) { console.error( "Erro na autenticação Microsoft Entra ID:", error ); diagnosticError(error); if (authUser) { authUser.innerHTML = "❌ Erro na validação da sessão."; } if (authStatus) { authStatus.style.borderColor = "#ef4444"; } return false; } }
